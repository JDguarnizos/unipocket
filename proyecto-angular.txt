import { Component, inject } from '@angular/core';
// ReactiveFormsModule habilita [formGroup] y formControlName en el HTML
import { ReactiveFormsModule, FormBuilder, Validators } from '@angular/forms';
import { Router } from '@angular/router';
import { LocalApi } from '../../../core/services/local-api';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [ReactiveFormsModule],
  templateUrl: './login.html',
  styleUrls: ['./login.css']
})
export class Login {
  // inject() trae los servicios de Angular (equivale a recibirlos en el constructor)
  private fb = inject(FormBuilder);
  private router = inject(Router);
  // RF13: los usuarios demostrativos vienen de la API local de Juan
  // (public/local-auth-api/usuarios.json y admins.json), SIN base de datos.
  private api = inject(LocalApi);

  // Mensaje que se muestra si las credenciales no coinciden (CU-2, curso alterno)
  mensajeError = '';

  // Formulario reactivo: cada campo con su valor inicial y sus validaciones
  loginForm = this.fb.group({
    correo: ['', [Validators.required, Validators.email]],
    contrasena: ['', [Validators.required, Validators.minLength(6)]]
  });

  // true si el campo es inválido Y el usuario ya lo tocó (así no salen errores apenas abre la página)
  campoInvalido(campo: string): boolean {
    const control = this.loginForm.get(campo);
    return !!control && control.invalid && control.touched;
  }

  // CU-1 / CU-2: se ejecuta al enviar el formulario
  iniciarSesion(): void {
    // Si hay errores, marcamos todos los campos como tocados para que se vean los mensajes
    if (this.loginForm.invalid) {
      this.loginForm.markAllAsTouched();
      return;
    }

    const { correo, contrasena } = this.loginForm.value;

    // Le pedimos al servicio que busque el correo y la contraseña en los JSON.
    // Como leer un archivo tarda, el resultado llega dentro de subscribe().
    this.api.validarCredenciales(correo ?? '', contrasena ?? '').subscribe({
      next: usuario => {
        if (usuario) {
          this.mensajeError = '';
          // Guardamos quién entró (con su rol: "usuario" o "administrador")
          this.api.iniciarSesion(usuario);
          // Por ahora ambos roles van al dashboard. Juan separará usuario/admin con los Guards.
          this.router.navigate(['/dashboard']);
        } else {
          this.mensajeError = 'Correo o contraseña incorrectos.';
        }
      },
      // Si no se pudo leer el JSON (ej. el archivo no existe)
      error: () => {
        this.mensajeError = 'No se pudo verificar el usuario. Intenta de nuevo.';
      }
    });
  }
}

import { Component, EventEmitter, HostListener, Input, OnInit, Output, inject } from '@angular/core';
import { ReactiveFormsModule, FormBuilder, Validators } from '@angular/forms';
import { Movimiento } from '../../../core/models/movimiento.model';

/*
  Modal "Registrar movimiento".
  - El padre (ej. el Dashboard) lo muestra con @if y le dice con qué tipo abrir.
  - Cuando el usuario guarda, el modal AVISA al padre con el evento (guardado).
  - Cuando cierra o cancela, avisa con el evento (cerrado).
*/
@Component({
  selector: 'app-registro-movimiento',
  standalone: true,
  imports: [ReactiveFormsModule],
  templateUrl: './registro-movimiento.html',
  styleUrls: ['./registro-movimiento.css']
})
export class RegistroMovimiento implements OnInit {
  private fb = inject(FormBuilder);

  // @Input: dato que recibe del padre. Con qué tipo se abre el modal.
  @Input() tipoInicial: 'gasto' | 'ingreso' = 'gasto';

  // @Output: eventos que el modal le envía al padre
  @Output() cerrado = new EventEmitter<void>();
  @Output() guardado = new EventEmitter<Movimiento>();

  // Categorías de gasto (coinciden con los colores --cat-* de styles.css)
  categoriasGasto = ['Alimentación', 'Transporte', 'Materiales e impresiones', 'Entretenimiento', 'Suscripciones', 'Otros'];
  categoriasIngreso = ['Mesada', 'Trabajo', 'Beca', 'Monitoría', 'Otros ingresos'];

  // Fecha de hoy en formato AAAA-MM-DD (el formato que usa el input type="date")
  hoy = new Date().toLocaleDateString('en-CA');

  // Formulario reactivo. La descripción es OPCIONAL (solo tiene límite de caracteres).
  movimientoForm = this.fb.group({
    tipo: ['gasto' as 'gasto' | 'ingreso', Validators.required],
    monto: [null as number | null, [Validators.required, Validators.min(1)]],
    categoria: ['', Validators.required],
    fecha: [this.hoy, Validators.required],
    descripcion: ['', Validators.maxLength(60)]
  });

  ngOnInit(): void {
    // Al abrir, seleccionamos el tipo que nos mandó el padre
    this.movimientoForm.patchValue({ tipo: this.tipoInicial });
  }

  // Atajo para saber si el tipo seleccionado es gasto
  get esGasto(): boolean {
    return this.movimientoForm.value.tipo === 'gasto';
  }

  // Lista de categorías según el tipo
  get categorias(): string[] {
    return this.esGasto ? this.categoriasGasto : this.categoriasIngreso;
  }

  // Cambia el tipo con los botones Gasto / Ingreso y limpia la categoría (las listas son distintas)
  seleccionarTipo(tipo: 'gasto' | 'ingreso'): void {
    this.movimientoForm.patchValue({ tipo, categoria: '' });
  }

  campoInvalido(campo: string): boolean {
    const control = this.movimientoForm.get(campo);
    return !!control && control.invalid && control.touched;
  }

  guardar(): void {
    if (this.movimientoForm.invalid) {
      this.movimientoForm.markAllAsTouched();
      return;
    }

    // Armamos el objeto Movimiento con los valores del formulario
    const valores = this.movimientoForm.getRawValue();
    const movimiento: Movimiento = {
      tipo: valores.tipo!,
      monto: Number(valores.monto),
      categoria: valores.categoria!,
      fecha: valores.fecha!,
      descripcion: valores.descripcion ?? ''
    };

    // Le enviamos el movimiento al padre. Él decide qué hacer (sumar/restar saldo, guardarlo en la API, etc.)
    this.guardado.emit(movimiento);
  }

  cerrar(): void {
    this.cerrado.emit();
  }

  // Permite cerrar el modal con la tecla Escape
  @HostListener('document:keydown.escape')
  alPresionarEscape(): void {
    this.cerrar();
  }
}

// Funciones de ayuda reutilizables para mostrar datos en pantalla :).
// Son funciones normales (no componentes ni servicios): se importan donde se necesiten.

// Convierte un numero a pesos colombianos con NumberFormat (herramienta nativa de JS (interesante)): 860000 -> "$860.000"
export function formatearPesos(valor: number): string {
  return '$' + valor.toLocaleString('es-CO');
}

// Versión corta para espacios pequeños: 340000 -> "$340k"
export function formatearPesosCorto(valor: number): string {
  return valor >= 1000 ? '$' + Math.round(valor / 1000) + 'k' : '$' + valor;
}

// Convierte "2026-11-12" -> "12 de nov de 2026"
// Se agrega 'T00:00' para que JavaScript use la hora local y no cambie el día.
export function formatearFecha(fecha: string): string {
  return new Date(fecha + 'T00:00').toLocaleDateString('es-CO', {
    day: 'numeric',
    month: 'short',
    year: 'numeric'
  });
}

// Color de cada categoría de gasto (variables --cat-* de styles.css).
// Si la categoría no está en la lista, usa el color de "Otros".
const COLORES_CATEGORIA: Record<string, string> = {
  'Alimentación': 'var(--cat-alimentacion)',
  'Transporte': 'var(--cat-transporte)',
  'Materiales e impresiones': 'var(--cat-materiales)',
  'Entretenimiento': 'var(--cat-entretenimiento)',
  'Suscripciones': 'var(--cat-suscripciones)'
};

export function colorCategoria(categoria: string): string {
  return COLORES_CATEGORIA[categoria] ?? 'var(--cat-otros)';
}

import { Component, OnInit, computed, inject } from '@angular/core';
import { ReactiveFormsModule, FormBuilder } from '@angular/forms';
// toSignal convierte los cambios de un formulario (Observable) en un signal
import { toSignal } from '@angular/core/rxjs-interop';
import { Router } from '@angular/router';

import { LocalApi } from '../../core/services/local-api';
import { Movimiento } from '../../core/models/movimiento.model';
import { formatearFecha, formatearPesos } from '../../core/utils/formato';
import { BotonGenericoComponent } from '../../shared/components/boton-generico/boton-generico';
import { ItemMovimientoComponent } from '../../shared/components/item-movimiento/item-movimiento';
import { DesgloseCategorias } from '../../shared/components/desglose-categorias/desglose-categorias';
import { RegistroMovimiento } from './registro-movimiento/registro-movimiento';

/*
  Vista "Historial de movimientos".
  - Muestra TODOS los movimientos del usuario, del más reciente al más antiguo.
  - Tiene filtros (búsqueda, tipo y categoría) hechos con un formulario reactivo.
  - Al lado muestra el desglose de gastos por categoría.
*/
@Component({
  selector: 'app-movimientos',
  standalone: true,
  imports: [ReactiveFormsModule, BotonGenericoComponent, ItemMovimientoComponent, DesgloseCategorias, RegistroMovimiento],
  templateUrl: './movimientos.html',
  styleUrls: ['./movimientos.css']
})
export class Movimientos implements OnInit {
  private api = inject(LocalApi);
  private router = inject(Router);
  private fb = inject(FormBuilder);

  // ===== Filtros (formulario reactivo) =====
  filtrosForm = this.fb.group({
    busqueda: [''],
    tipo: ['todos'],   // 'todos' | 'ingreso' | 'gasto'
    categoria: ['']    // '' = todas las categorías
  });

  // Cada vez que el usuario escribe o cambia un filtro, este signal se actualiza.
  // initialValue: el valor que tiene antes de que el usuario toque algo.
  filtros = toSignal(this.filtrosForm.valueChanges, { initialValue: this.filtrosForm.value });

  // ===== Datos =====

  // Todos los movimientos del usuario, ordenados por fecha (más reciente primero).
  // localeCompare compara textos: como la fecha es "AAAA-MM-DD", el orden alfabético = orden por fecha.
  misMovimientos = computed(() => {
    const usuario = this.api.usuarioActual();
    return this.api.movimientos()
      .filter(m => m.idUsuario === usuario?.id)
      .sort((a, b) => b.fecha.localeCompare(a.fecha));
  });

  // Lista de categorías que el usuario ha usado (para el select del filtro), sin repetidas
  categoriasDisponibles = computed(() =>
    [...new Set(this.misMovimientos().map(m => m.categoria))].sort()
  );

  // Movimientos que cumplen TODOS los filtros
  movimientosFiltrados = computed(() => {
    const f = this.filtros();
    const texto = (f.busqueda ?? '').trim().toLowerCase();

    return this.misMovimientos().filter(m => {
      const cumpleTipo = f.tipo === 'todos' || m.tipo === f.tipo;
      const cumpleCategoria = !f.categoria || m.categoria === f.categoria;
      const cumpleTexto = !texto
        || m.descripcion.toLowerCase().includes(texto)
        || m.categoria.toLowerCase().includes(texto);
      return cumpleTipo && cumpleCategoria && cumpleTexto;
    });
  });

  // Totales de lo que se está viendo (cambian con los filtros)
  totalIngresos = computed(() => this.sumar(this.movimientosFiltrados(), 'ingreso'));
  totalGastos = computed(() => this.sumar(this.movimientosFiltrados(), 'gasto'));

  // ===== Modal "Registrar movimiento" (el mismo del Dashboard) =====
  modalAbierto = false;
  tipoModal: 'gasto' | 'ingreso' = 'gasto';

  ngOnInit(): void {
    // TEMPORAL (igual que en el Dashboard): sin sesión, volvemos al login.
    // Cuando estén los Guards de Juan, esta validación pasa a ellos.
    if (!this.api.usuarioActual()) {
      this.router.navigate(['/login']);
      return;
    }
    this.api.cargarDatosFinancieros();
  }

  // Suma los montos de un tipo (ingreso o gasto)
  private sumar(lista: Movimiento[], tipo: 'ingreso' | 'gasto'): number {
    return lista.filter(m => m.tipo === tipo).reduce((suma, m) => suma + m.monto, 0);
  }

  limpiarFiltros(): void {
    this.filtrosForm.reset({ busqueda: '', tipo: 'todos', categoria: '' });
  }

  abrirModal(tipo: 'gasto' | 'ingreso'): void {
    this.tipoModal = tipo;
    this.modalAbierto = true;
  }

  cerrarModal(): void {
    this.modalAbierto = false;
  }

  // El modal emite (guardado): el servicio lo guarda y TODAS las listas se recalculan solas
  registrarMovimiento(movimiento: Movimiento): void {
    const usuario = this.api.usuarioActual();
    if (!usuario) return;
    this.api.registrarMovimiento(usuario.id, movimiento);
    this.cerrarModal();
  }

  // Funciones de formato para usarlas en el HTML
  formatearPesos = formatearPesos;
  formatearFecha = formatearFecha;
}

import { Component, computed, input } from '@angular/core';
import { Movimiento } from '../../../core/models/movimiento.model';
import { colorCategoria, formatearPesos, formatearPesosCorto } from '../../../core/utils/formato';

// Forma de cada fila del desglose
interface CategoriaDesglose {
  nombre: string;
  monto: number;
  porcentaje: number; // redondeado, para mostrar en pantalla
  color: string;
}

/*
  Desglose visual de los gastos por categoría (es la gráfica de dona + la lista).
  Es un componente COMPARTIDO: recibe una lista de movimientos y calcula todo solo.
  Se puede usar en el Historial, en el Dashboard o en Reportes :).

  Nota: Uso:  <app-desglose-categorias [movimientos]="misMovimientos()" />
*/
@Component({
  selector: 'app-desglose-categorias',
  standalone: true,
  imports: [],
  templateUrl: './desglose-categorias.html',
  styleUrls: ['./desglose-categorias.css']
})
export class DesgloseCategorias {
  // input() es la versión "signal" de @Input().
  // La usamos porque así los computed() de abajo se recalculan solos cuando cambia la lista.
  movimientos = input<Movimiento[]>([]);

  // 1) Solo nos interesan los gastos
  private gastos = computed(() => this.movimientos().filter(m => m.tipo === 'gasto'));

  // 2) Total gastado (se muestra en el centro de la dona)
  totalGastos = computed(() => this.gastos().reduce((suma, m) => suma + m.monto, 0));

  // 3) Agrupamos por categoría y calculamos el porcentaje de cada una
  categorias = computed<CategoriaDesglose[]>(() => {
    const total = this.totalGastos();

    // Objeto acumulador: { "Alimentación": 119000, "Transporte": 68000, ... }
    const totalesPorCategoria: Record<string, number> = {};
    for (const gasto of this.gastos()) {
      totalesPorCategoria[gasto.categoria] = (totalesPorCategoria[gasto.categoria] ?? 0) + gasto.monto;
    }

    // Convertimos el objeto en una lista y la ordenamos de mayor a menor gasto
    return Object.entries(totalesPorCategoria)
      .map(([nombre, monto]) => ({
        nombre,
        monto,
        porcentaje: total > 0 ? Math.round((monto / total) * 100) : 0,
        color: colorCategoria(nombre)
      }))
      .sort((a, b) => b.monto - a.monto);
  });

  // 4) Fondo de la dona: un conic-gradient (degradado en círculo) con un tramo por categoría.
  //    Ej: "conic-gradient(amarillo 0% 35%, azul 35% 55%, ...)"
  gradienteDona = computed(() => {
    const total = this.totalGastos();
    if (total === 0) return 'var(--neutral-200)';

    let acumulado = 0; // porcentaje donde empieza cada tramo
    const tramos = this.categorias().map(cat => {
      const inicio = acumulado;
      acumulado += (cat.monto / total) * 100;
      return `${cat.color} ${inicio}% ${acumulado}%`;
    });
    return `conic-gradient(${tramos.join(', ')})`;
  });

  // Funciones de formato para usarlas en el HTML
  formatearPesos = formatearPesos;
  formatearPesosCorto = formatearPesosCorto;
}

import { Component, signal, HostListener, ElementRef, Output, EventEmitter } from '@angular/core';
import { CommonModule } from '@angular/common';

// Interfaz que define cómo se ve cada notificación
interface Notificacion {
  id: number;
  mensaje: string;    // Texto principal de la notificación
  tiempo: string;     // Cuándo ocurrió (ej: "Hace 5 min")
  leida: boolean;     // Si el usuario ya la vio o no
}

@Component({
  imports: [CommonModule],
  selector: 'app-header',
  styleUrl: './header.css',
  templateUrl: './header.html',
})
export class Header {

  //Abrir y cerrar el menu lateral en los dispositivos moviles
  @Output() toggleMenu = new EventEmitter<void>();

  onToggleMenu(): void {
    this.toggleMenu.emit();
  }

  // ── Datos del usuario ──────────────────────────────────────────────────────
  nombreUsuario: string = 'Estudiante';
  rolUsuario: string = 'Ucompensar';

  // ── Estado del menú de notificaciones ─────────────────────────────────────
  // signal() es la forma moderna de Angular para variables reactivas.
  // Cuando cambia, el HTML se actualiza automáticamente.
  mostrarNotificaciones = signal(false);

  // ── Lista de notificaciones de ejemplo ────────────────────────────────────
  notificaciones: Notificacion[] = [
    {
      id: 1,
      mensaje: 'Llevas el 68% de tu presupuesto mensual.',
      tiempo: 'Hace 5 min',
      leida: false,
    },
    {
      id: 2,
      mensaje: 'Meta "Viaje de semestre" al 45% 🎯',
      tiempo: 'Hace 1 hora',
      leida: false,
    },
    {
      id: 3,
      mensaje: 'Se registró un ingreso de $150.000.',
      tiempo: 'Ayer, 16:00',
      leida: true,
    },
  ];

  // Calculamos cuántas notificaciones NO han sido leídas
  get cantidadSinLeer(): number {
    return this.notificaciones.filter(n => !n.leida).length;
  }

  // ── Constructor con inyección del elemento raíz ───────────────────────────
  // Necesitamos la referencia al elemento para detectar clics fuera del menú
  constructor(private elementRef: ElementRef) {}

  // ── Abrir / cerrar el panel de notificaciones ─────────────────────────────
  toggleNotificaciones(): void {
    this.mostrarNotificaciones.update(valor => !valor);
  }

  // ── Marcar una notificación como leída ────────────────────────────────────
  marcarComoLeida(id: number): void {
    const notif = this.notificaciones.find(n => n.id === id);
    if (notif) {
      notif.leida = true;
    }
  }

  // ── Marcar todas como leídas ──────────────────────────────────────────────
  marcarTodasComoLeidas(): void {
    this.notificaciones.forEach(n => (n.leida = true));
  }

  // ── Cerrar el panel si el usuario hace clic fuera del header ─────────────
  // @HostListener escucha eventos en toda la página (document)
  @HostListener('document:click', ['$event'])
  cerrarAlClickearAfuera(evento: MouseEvent): void {
    // Si el clic ocurrió FUERA de este componente, cerramos el panel
    const dentroDelComponente = this.elementRef.nativeElement.contains(evento.target);
    if (!dentroDelComponente) {
      this.mostrarNotificaciones.set(false);
    }
  }
}

import { Component, OnInit, computed, inject } from '@angular/core';
import { CommonModule } from '@angular/common';
import { Router } from '@angular/router';

import { BotonGenericoComponent } from '../../shared/components/boton-generico/boton-generico';
import { TarjetaResumenComponent } from '../../shared/components/tarjeta-resumen/tarjeta-resumen';
import { ItemMovimientoComponent } from '../../shared/components/item-movimiento/item-movimiento';
import { RegistroMovimiento } from '../movimientos/registro-movimiento/registro-movimiento';
import { Movimiento } from '../../core/models/movimiento.model';
import { LocalApi } from '../../core/services/local-api';

@Component({
  selector: 'app-dashboard',
  standalone: true,
  imports: [CommonModule, BotonGenericoComponent, TarjetaResumenComponent, ItemMovimientoComponent, RegistroMovimiento],
  templateUrl: './dashboard.html',
  styleUrls: ['./dashboard.css']
})
export class Dashboard implements OnInit {
  private api = inject(LocalApi);
  private router = inject(Router);

  modalAbierto = false;
  tipoModal: 'gasto' | 'ingreso' = 'gasto';

  // ===== Datos conectados a la API local =====
  usuario = this.api.usuarioActual;

  // 1) El registro financiero del usuario. Lo buscamos una sola vez y las demás tarjetas salen de aquí.
registroActual = computed(() => {
    const usuario = this.api.usuarioActual();
    if (!usuario) return undefined;
  // Buscamos haciendo match estricto o flexible con el id del usuario
  return this.api.registrosFinancieros().find(r => r.idUsuario == usuario.id);
  });
  // 2) Dinero disponible = lo que ha entrado - lo que ha salido
  saldoDisponible = computed(() => {
    const r = this.registroActual();
    return r ? r.totalIngresos - r.totalGastos : 0;
  });

  // 3) Porcentaje del presupuesto que ya se gastó
  porcentajePresupuesto = computed(() => {
    const r = this.registroActual();
    if (!r || r.presupuesto === 0) return 0;
    return Math.round((r.totalGastos / r.presupuesto) * 100);
  });

  // 4) Presupuesto restante (puede ser negativo si se pasó del límite)
  presupuestoRestante = computed(() => {
    const r = this.registroActual();
    return r ? r.presupuesto - r.totalGastos : 0;
  });

  // 5) Días que faltan para terminar el mes (contando hoy)
  diasRestantes = computed(() => {
    const hoy = new Date();
    const ultimoDia = new Date(hoy.getFullYear(), hoy.getMonth() + 1, 0).getDate();
    return ultimoDia - hoy.getDate() + 1;
  });

  // 6) Gasto diario recomendado = lo que queda del presupuesto / días que faltan
  gastoDiarioRecomendado = computed(() =>
    Math.round(Math.max(0, this.presupuestoRestante()) / this.diasRestantes())
  );

  // Mes actual para la cabecera: "noviembre de 2026"
  mesActual = new Date().toLocaleDateString('es-co', { month: 'long', year: 'numeric' });

  // Movimientos del usuario que inició sesión
  movimientosRecientes = computed(() => {
    const usuario = this.api.usuarioActual();
    return this.api.movimientos()
      .filter(m => m.idUsuario === usuario?.id)
      .slice(0, 5)
      .map(m => ({
        titulo: m.descripcion || m.categoria,
        subtitulo: `${m.categoria} - ${m.fecha}`,
        monto: m.monto,
        tipo: m.tipo
      }));
  });

  ngOnInit(): void {
    if (!this.api.usuarioActual()) {
      this.router.navigate(['/login']);
      return;
    }
    // Lee los JSON solo la primera vez gracias a la bandera restaurada
    this.api.cargarDatosFinancieros();
  }

  abrirModal(tipo: 'gasto' | 'ingreso'): void {
    this.tipoModal = tipo;
    this.modalAbierto = true;
  }

  cerrarModal(): void {
    this.modalAbierto = false;
  }

  registrarMovimiento(movimiento: Movimiento): void {
    const usuario = this.api.usuarioActual();
    if (!usuario) return;

    this.api.registrarMovimiento(usuario.id, movimiento);
    this.cerrarModal();
  }

  formatearPesos(valor: number): string {
    return '$' + Number(valor || 0).toLocaleString('es-CO');
  }
}

import { inject, Injectable, Service, signal } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable, catchError, forkJoin, map, of, shareReplay, tap, throwError } from 'rxjs';
import { Admin } from '../models/admin.model';
import { Estadisticas } from '../models/estadisticas.model';
import { MetaAhorro } from '../models/meta_ahorro.model';
import { Notificacion } from '../models/notificacion.model';
import { RegistroFinanciero } from '../models/reg_financiero.model';
import { Usuario } from '../models/usuario.model';
import { Movimiento, MovimientoUsuario } from '../models/movimiento.model';

@Service()
export class LocalApi {
  private http = inject(HttpClient);

  // =====================================================================
  // 1. LECTURA DE LOS JSON
  // Cada método lee un archivo de public/local-auth-api/ con HttpClient
  // y devuelve un Observable (los datos llegan cuando alguien hace .subscribe()).
  // =====================================================================

  obtenerAdmins(): Observable<Admin[]> {
    return this.http.get<Admin[]>('local-auth-api/admins.json');
  }
  obtenerEstadisticas(): Observable<Estadisticas[]> {
    return this.http.get<Estadisticas[]>('local-auth-api/estadisticas.json');
  }
  // Corregido (Nicolás): antes usaba como tipo el componente MetasAhorro; el tipo correcto es el modelo Meta
  // Cambiado (Guarniz): Reflejado el cambio de nombre a MetaAhorro.
  obtenerMetasAhorro(): Observable<MetaAhorro[]> {
    return this.http.get<MetaAhorro[]>('local-auth-api/metas.json');
  }
  obtenerNotificaciones(): Observable<Notificacion[]> {
    return this.http.get<Notificacion[]>('local-auth-api/notificaciones.json');
  }
  obtenerRegistrosFinancieros(): Observable<RegistroFinanciero[]> {
    return this.http.get<RegistroFinanciero[]>('local-auth-api/registrosFinancieros.json');
  }
  // Corregido (Nicolás): antes leía admins.json en vez de usuarios.json
  obtenerUsuarios(): Observable<Usuario[]> {
    return this.http.get<Usuario[]>('local-auth-api/usuarios.json');
  }
  // Nuevo (Nicolás): movimientos (ingresos y gastos) de todos los usuarios
  obtenerMovimientos(): Observable<MovimientoUsuario[]> {
    return this.http.get<MovimientoUsuario[]>('local-auth-api/movimientos.json');
  }

  // =====================================================================
  // 2. LOGIN
  // =====================================================================

  // Usuario que inició sesión. null = nadie ha iniciado sesión.
  // Es un signal: cualquier componente que lo lea se actualiza solo cuando cambia.
  usuarioActual = signal<Usuario | null>(null);

  // Busca un usuario o admin con ese correo y contraseña.
  // forkJoin espera a que lleguen LOS DOS archivos (usuarios y admins) y luego
  // map une las dos listas y busca la coincidencia. Devuelve undefined si no existe.
  validarCredenciales(correo: string, contrasena: string): Observable<Usuario | undefined> {
    return forkJoin([this.obtenerUsuarios(), this.obtenerAdmins()]).pipe(
      map(([usuarios, admins]) =>
        [...usuarios, ...admins].find((u) => u.correo === correo && u.contrasena === contrasena),
      ),
    );
  }

  // Guarda en memoria quién inició sesión (incluye su rol: "usuario" o "administrador")
  iniciarSesion(usuario: Usuario): void {
    this.usuarioActual.set(usuario);
  }

  // =====================================================================
  // 3. DATOS FINANCIEROS EN MEMORIA
  // Los .json de public/ son de SOLO LECTURA (HttpClient no puede escribir en ellos).
  // Por eso: los leemos UNA vez, los copiamos en signals y desde ahí los modificamos.
  // Como el servicio es único en toda la app, los cambios se mantienen al navegar
  // entre vistas, pero se pierden al recargar la página (vuelve a leer el JSON).
  // =====================================================================

  movimientos = signal<MovimientoUsuario[]>([]);
  registrosFinancieros = signal<RegistroFinanciero[]>([]);
  private datosFinancierosCargados = false;

  // Lee movimientos.json y registrosFinancieros.json y llena los signals.
  // Si ya se cargaron antes, no hace nada (así no se borran los movimientos nuevos).
  cargarDatosFinancieros(): void {
    if (this.datosFinancierosCargados) return;
    this.datosFinancierosCargados = true;

    forkJoin([this.obtenerMovimientos(), this.obtenerRegistrosFinancieros()]).subscribe({
      next: ([movimientos, registros]) => {
        this.movimientos.set(movimientos);
        this.registrosFinancieros.set(registros);
      },
      error: () => {
        // Si falla la lectura, permitimos intentarlo de nuevo la próxima vez
        this.datosFinancierosCargados = false;
        console.error('No se pudieron cargar los datos financieros');
      },
    });
  }

  // Registra un movimiento nuevo del usuario indicado.
  registrarMovimiento(idUsuario: number, movimiento: Movimiento): void {
    // 1) Agregamos el movimiento de PRIMERO en la lista (es el más reciente).
    //    update() recibe la lista actual y devuelve una lista nueva.
    const nuevo: MovimientoUsuario = { ...movimiento, idUsuario };
    this.movimientos.update((lista) => [nuevo, ...lista]);

    // 2) Actualizamos los totales del registro financiero de ese usuario:
    //    un ingreso suma a totalIngresos y un gasto suma a totalGastos.
    //    (El saldo disponible = totalIngresos - totalGastos, así que el ingreso
    //    lo hace subir y el gasto lo hace bajar.)
    this.registrosFinancieros.update((registros) =>
      registros.map((registro) => {
        if (registro.idUsuario !== idUsuario) return registro; // los demás quedan igual
        return movimiento.tipo === 'ingreso'
          ? { ...registro, totalIngresos: registro.totalIngresos + movimiento.monto }
          : { ...registro, totalGastos: registro.totalGastos + movimiento.monto };
      }),
    );
  }

  // Actualización de datos dentro de metas de ahorro, hecho con las manos.
  // La idea es usar la misma forma que se usa arriba para cargar datos, manipularlos en memoria
  // Dado que angular por defecto no puede modificar JSONs

  metasAhorro = signal<MetaAhorro[]>([]);
  private metasAhorroCargadas = false;
  private metasAhorroCarga: Observable<MetaAhorro[]> | null = null;

  // Mismo flujo, obtener datos suscribiendose al observable, si falla notificando en consola
  cargarMetasAhorro(): Observable<MetaAhorro[]> {
    if (this.metasAhorroCargadas) return of(this.metasAhorro());

    if (!this.metasAhorroCarga) {
      this.metasAhorroCarga = this.obtenerMetasAhorro().pipe(
        tap((metasAhorro) => {
          this.metasAhorro.set(metasAhorro);
          this.metasAhorroCargadas = true;
        }),
        catchError((error) => {
          this.metasAhorroCarga = null;
          console.error('La carga de MetasAhorro ha fallado.');
          return throwError(() => error);
        }),
        shareReplay({ bufferSize: 1, refCount: false }),
      );
    }

    return this.metasAhorroCarga;
  }

  // Carga usuarios, verifica que el usuario dentro de la meta de ahorro exista y inserta la meta en tal caso
  registrarMetaAhorro(nuevaMetaAhorro: MetaAhorro): void {
    // Cargar la lista de usuarios.
    // Podría dejar esto en una función nueva pero no se puede suscribir a la misma para hacer los cambios, dado que se desincroniza
    // y puede saltarse uno de los usuarios.
    var listaUsuarios = signal<Usuario[]>([]);
    this.obtenerUsuarios().subscribe({
      next: (Usuarios) => {
        listaUsuarios.set(Usuarios);
        var usuarioExiste = false;

        // Es la manera que encontré para iterar en esta lista xd, en general no hace cambios en la lista de usuarios
        listaUsuarios.update((usuarios) =>
          usuarios.map((usuario) => {
            if (nuevaMetaAhorro.idUsuario == usuario.id) {
              usuarioExiste = true;
            }
            return usuario;
          }),
        );

        if (usuarioExiste)
          this.metasAhorro.update((MetasActuales) => [nuevaMetaAhorro, ...MetasActuales]);
      },
      error: () => {
        console.error('La carga de ListaUsuarios ha fallado durante el registro de MetaAhorro.');
      },
    });
  }

  // Verifica que el aporte no sea 0, que haya expirado, verifica nombre y id del usuario con la lista actual de metas, añade los aportes
  registrarAporteMetaAhorro(
    idUsuario: number,
    nombreMeta: string,
    aporte: number,
    fechaAporte: Date = new Date(),
  ): void {
    if (aporte <= 0) return;
    const now = new Date();
    if (fechaAporte < now) return;
    this.cargarMetasAhorro().subscribe({
      next: () =>
        this.metasAhorro.update((metas) =>
          metas.map((meta) => {
            // Si la meta no concuerda con el idUsuario y nombre, no hace NADA.
            if (meta.idUsuario !== idUsuario) return meta;
            if (meta.nombre.toLowerCase() !== nombreMeta.toLowerCase()) return meta;

            const listaAportes = [
              ...meta.listaAportes,
              { monto: aporte, fechaAporte: fechaAporte.toISOString() },
            ];
            const totalAportes = listaAportes.reduce((total, valor) => total + valor.monto, 0);

            return {
              ...meta,
              listaAportes,
              completado: meta.completado || totalAportes >= meta.valorObjetivo,
            };
          }),
        ),
      error: () => console.error('No se pudo registrar el aporte a la meta.'),
    });
  }


  // Devuelve el progreso de meta de ahorro en porcentaje, pide id y nombre meta buscar y calcular esa meta. 
  // Esto debe ser totalAportado, o lo que sea que le dijo claude a nico, no voy a modificar las interfaces y json para añadir un total que se puede calcular tan facil como un for each
  calcularProgresoMetaAhorro(idUsuario: number, nombreMeta: string): Observable<number> {
    return this.cargarMetasAhorro().pipe(
      map((metas) => {
        const meta = metas.find(
          (item) =>
            item.idUsuario === idUsuario &&
            item.nombre.toLowerCase() === nombreMeta.toLowerCase(),
        );
        if (!meta || meta.valorObjetivo <= 0) return 0;

        const totalAportes = meta.listaAportes.reduce((total, aporte) => total + aporte.monto, 0);
        return (totalAportes / meta.valorObjetivo) * 100;
      }),
    );
  }
}
